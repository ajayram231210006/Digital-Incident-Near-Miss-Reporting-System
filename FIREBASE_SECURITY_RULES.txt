// Firebase Realtime Database Security Rules
// These rules enforce data security and access control

// Replace your current Firebase rules with this content
// Go to Firebase Console → Your Project → Realtime Database → Rules

{
  "rules": {
    "users": {
      "$uid": {
        // Only the user themselves or an admin can read their data
        ".read": "auth.uid === $uid || root.child('users').child(auth.uid).child('role').val() === 'admin'",
        
        // Only the user can write to their own data during signup
        // But after account creation, only admins can modify status/role
        ".write": "!data.exists() || auth.uid === $uid || root.child('users').child(auth.uid).child('role').val() === 'admin'",
        
        "role": {
          // Admins can modify role
          ".write": "root.child('users').child(auth.uid).child('role').val() === 'admin'"
        },
        
        "status": {
          // Admins can modify status (for approvals/rejections)
          ".write": "root.child('users').child(auth.uid).child('role').val() === 'admin'"
        }
      }
    },
    
    "reports": {
      "$reportId": {
        ".read": "root.child('users').child(auth.uid).exists()",
        ".write": "root.child('users').child(auth.uid).exists()"
      }
    }
  }
}

/*
RULES EXPLANATION:

1. users/$uid/.read
   - Users can only read their own profile
   - Admins can read any user's profile

2. users/$uid/.write
   - During signup (no data exists): new users can create their own profile
   - After signup: users can only modify their own data (except role/status)
   - Admins: can modify any field including role and status

3. users/$uid/role
   - Only admins can change this field

4. users/$uid/status
   - Only admins can change this field (for approvals)

5. reports
   - Any authenticated user can read and write reports
   - (You may want to tighten this based on your needs)
*/
